cmake_minimum_required(VERSION 3.28)
project(DeckerMemoryTraceInspector LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(DK_TRACY_SOURCE_DIR "" CACHE PATH "Matching Tracy sources prepared by the project vcpkg overlay")
if(NOT EXISTS "${DK_TRACY_SOURCE_DIR}/cmake/server.cmake")
    message(FATAL_ERROR "Set DK_TRACY_SOURCE_DIR to the vcpkg-patched Tracy 0.14.1 source directory.")
endif()
include("${DK_TRACY_SOURCE_DIR}/cmake/version.cmake")
if(NOT TRACY_VERSION_STRING STREQUAL "0.14.1")
    message(FATAL_ERROR "Memory capture inspection requires Tracy 0.14.1.")
endif()
find_package(capstone CONFIG REQUIRED)
find_package(zstd CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(PPQSort CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
set(NO_STATISTICS OFF)
include("${DK_TRACY_SOURCE_DIR}/cmake/server.cmake")
if(WIN32)
    target_compile_definitions(TracyServer PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
endif()
if(MSVC)
    target_compile_options(TracyServer PUBLIC /utf-8)
endif()
add_executable(dk_memory_trace_inspect MemoryTraceInspector.cpp)
set_target_properties(dk_memory_trace_inspect PROPERTIES OUTPUT_NAME "dk-memory-trace-inspect")
target_link_libraries(dk_memory_trace_inspect PRIVATE TracyServer nlohmann_json::nlohmann_json)
