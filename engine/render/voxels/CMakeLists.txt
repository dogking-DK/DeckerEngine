set(CMAKE_FOLDER "Engine/Render/Voxels")
foreach(dependency dk_geometry_voxels_sparse dk_graphics_device dk_graphics_graph dk_graphics_shaders)
    if(NOT TARGET ${dependency})
        message(FATAL_ERROR "DK_BUILD_RENDER_VOXELS requires ${dependency}; enable Voxel VDB, Device, Graph and Shaders")
    endif()
endforeach()
add_library(dk_render_voxels STATIC src/VoxelGpu.cpp)
add_library(dk::render_voxels ALIAS dk_render_voxels)
target_sources(dk_render_voxels PUBLIC FILE_SET HEADERS BASE_DIRS include FILES include/dk/render/VoxelGpu.hpp)
target_link_libraries(dk_render_voxels PUBLIC dk::geometry_voxels_sparse dk::graphics_device PRIVATE dk::graphics_graph dk::graphics_shaders)
target_compile_definitions(dk_render_voxels PRIVATE DK_VOXEL_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders" DK_VOXEL_NANO_INCLUDE_DIR="${DK_NANOVDB_INCLUDE_DIR}")
dk_target_defaults(dk_render_voxels)
