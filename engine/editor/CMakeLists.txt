set(CMAKE_FOLDER "Engine/Editor")
add_library(dk_editor_model STATIC src/Workspace.cpp src/Camera.cpp src/TransformEdit.cpp)
add_library(dk::editor_model ALIAS dk_editor_model)
target_sources(dk_editor_model PUBLIC FILE_SET HEADERS BASE_DIRS include FILES include/dk/editor/Workspace.hpp include/dk/editor/Camera.hpp include/dk/editor/TransformEdit.hpp)
target_link_libraries(dk_editor_model PUBLIC dk::runtime dk::geometry)
if(WIN32)
    target_link_libraries(dk_editor_model PUBLIC dk::automation_server)
endif()
dk_target_defaults(dk_editor_model)

if(DK_BUILD_EDITOR)
    if(NOT TARGET dk_graphics_presentation OR NOT TARGET dk_render_disk OR NOT TARGET dk_render_pipeline)
        message(FATAL_ERROR "DK_BUILD_EDITOR requires Presentation, Render Disk and Render Pipeline")
    endif()
    find_package(imgui CONFIG REQUIRED)
    find_package(SDL3 CONFIG REQUIRED)
    add_library(dk_editor STATIC src/Workbench.cpp src/WorkbenchUi.cpp src/Viewport.cpp src/ViewportInput.cpp src/GuiRenderer.cpp src/SmokeDriver.cpp)
    add_library(dk::editor ALIAS dk_editor)
    target_sources(dk_editor PUBLIC FILE_SET HEADERS BASE_DIRS include FILES include/dk/editor/Workbench.hpp)
    target_link_libraries(dk_editor PRIVATE dk::editor_model dk::graphics_presentation dk::render_disk dk::render_pipeline imgui::imgui SDL3::SDL3)
    target_compile_definitions(dk_editor PRIVATE DK_EDITOR_SHADER_DIR="${PROJECT_SOURCE_DIR}/shaders/render")
    dk_target_defaults(dk_editor)
endif()
