if(NOT DK_BUILD_MEMORY)
    message(FATAL_ERROR "DK_BUILD_GRAPHICS_DEVICE requires DK_BUILD_MEMORY")
endif()
find_package(VulkanHeaders 1.4 CONFIG REQUIRED)
find_package(volk CONFIG REQUIRED)
find_package(vk-bootstrap CONFIG REQUIRED)
find_package(VulkanMemoryAllocator CONFIG REQUIRED)
# Isolate third-party implementation warnings from /W4 /WX on engine sources.
add_library(dk_graphics_vma STATIC src/Vma.cpp)
target_compile_features(dk_graphics_vma PRIVATE cxx_std_23)
target_compile_definitions(dk_graphics_vma PRIVATE VK_NO_PROTOTYPES)
target_link_libraries(dk_graphics_vma PRIVATE Vulkan::Headers GPUOpen::VulkanMemoryAllocator)
if(MSVC)
    target_compile_options(dk_graphics_vma PRIVATE /W0 /utf-8)
else()
    target_compile_options(dk_graphics_vma PRIVATE -w)
endif()
add_library(dk_graphics_device STATIC src/Device.cpp src/Selection.cpp src/Bootstrap.cpp src/Resources.cpp)
add_library(dk::graphics_device ALIAS dk_graphics_device)
target_sources(dk_graphics_device PUBLIC FILE_SET HEADERS BASE_DIRS include FILES
    include/dk/graphics/Device.hpp include/dk/graphics/Resources.hpp)
target_compile_definitions(dk_graphics_device PUBLIC VK_NO_PROTOTYPES
    VULKAN_HPP_ENABLE_DYNAMIC_LOADER_TOOL=0 VULKAN_HPP_NO_DEFAULT_DISPATCHER)
target_link_libraries(dk_graphics_device PUBLIC dk::core dk::memory Vulkan::Headers GPUOpen::VulkanMemoryAllocator
    PRIVATE dk::profiling volk::volk vk-bootstrap::vk-bootstrap dk_graphics_vma ${CMAKE_DL_LIBS})
dk_target_defaults(dk_graphics_device)
