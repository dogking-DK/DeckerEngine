add_executable(dk_run src/main.cpp)
set_target_properties(dk_run PROPERTIES OUTPUT_NAME "dk-run")
target_link_libraries(dk_run PRIVATE dk::core dk::profiling)
if(TARGET dk_runtime)
    target_link_libraries(dk_run PRIVATE dk::automation_server)
    target_compile_definitions(dk_run PRIVATE DK_RUN_WITH_RUNTIME=1)
endif()
dk_target_defaults(dk_run)

if(TARGET dk_scripting_luau)
    target_sources(dk_run PRIVATE src/ScriptRunner.cpp src/ScriptRunner.hpp)
    target_link_libraries(dk_run PRIVATE dk::scripting_luau)
    target_compile_definitions(dk_run PRIVATE DK_RUN_WITH_LUAU=1)
endif()

if(TARGET dk_render_services OR TARGET dk_render_simulation)
    dk_deploy_slang(dk_run)
endif()
